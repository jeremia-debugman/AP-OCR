  <!-- Custom In-App Toast -->
  <div id="customToast" style="display:none; position:fixed; bottom:80px; left:50%; transform:translateX(-50%); background:#323232; color:#fff; padding:10px 20px; border-radius:20px; font-size:13px; z-index:9999; box-shadow:0 4px 6px rgba(0,0,0,0.1); text-align:center;"></div>
  
  <script>
    let rawInvoices = [];
    let filteredInvoices = [];
    let currentIndex = 0;
    let decisions = {};

    window.onload = function() {
      google.script.run
        .withSuccessHandler(onDataLoaded)
        .withFailureHandler(err => {
          document.getElementById('counterText').innerText = 'Error loading queue.';
          alert('Server Error: ' + err);
        })
        .getInvoicesForReview();
    };

    function onDataLoaded(res) {
      if (Array.isArray(res)) {
        rawInvoices = res;
      } else if (res && Array.isArray(res.invoices)) {
        rawInvoices = res.invoices;
      } else {
        rawInvoices = [];
      }

      if (rawInvoices.length === 0) {
        document.getElementById('counterText').innerText = 'No pending invoices found in database.';
        return;
      }

      const hasChecked = rawInvoices.some(i => i.isSelectedInQueue === true);
      if (hasChecked) {
        setScope('selected');
      } else {
        setScope('all');
      }
    }

    function setScope(scope) {
      document.getElementById('scopeSelectedBtn').classList.toggle('active', scope === 'selected');
      document.getElementById('scopeAllBtn').classList.toggle('active', scope === 'all');

      if (scope === 'selected') {
        filteredInvoices = rawInvoices.filter(i => i.isSelectedInQueue === true);
        if (filteredInvoices.length === 0) {
          filteredInvoices = [...rawInvoices];
          document.getElementById('scopeSelectedBtn').classList.remove('active');
          document.getElementById('scopeAllBtn').classList.add('active');
        }
      } else {
        filteredInvoices = [...rawInvoices];
      }

      currentIndex = 0;
      renderCurrentInvoice();
    }

    function renderCurrentInvoice() {
      if (!filteredInvoices || filteredInvoices.length === 0) {
        document.getElementById('counterText').innerText = 'No invoices to display.';
        document.getElementById('contentContainer').style.display = 'none';
        document.getElementById('statusBadge').style.display = 'none';
        return;
      }

      document.getElementById('contentContainer').style.display = 'block';
      const inv = filteredInvoices[currentIndex];
      document.getElementById('counterText').innerText = `Invoice ${currentIndex + 1} of ${filteredInvoices.length}`;

      const dictKey = inv.invoiceNumber ? inv.invoiceNumber : ('row_' + inv.invRow);
      const decision = decisions[dictKey];
      const badge = document.getElementById('statusBadge');
      
      badge.style.display = 'block';
      if (decision && decision.action === 'Approve') {
        badge.innerText = 'Approve';
        badge.style.background = '#e6f4ea'; badge.style.color = '#1e8e3e';
      } else if (decision && decision.action === 'Reject') {
        badge.innerText = 'Reject';
        badge.style.background = '#fce8e6'; badge.style.color = '#d93025';
      } else {
        badge.innerText = 'Pending';
        badge.style.background = '#f1f3f4'; badge.style.color = '#5f6368';
      }

      // Populate Header
      document.getElementById('invNumber').value = inv.invoiceNumber || '(No Invoice Number)';
      document.getElementById('rawVendor').value = inv.vendorName || '';
      document.getElementById('invDate').value = inv.invoiceDate || '';

      // SLA Warning Badge Logic
      const slaBadge = document.getElementById('slaBadge');
      if (inv.slaStatus === 'Exceeded SLA') {
        slaBadge.innerText = `EXCEEDED (${inv.processingDays} DAYS)`;
        slaBadge.style.background = '#fce8e6'; // Light Red
        slaBadge.style.color = '#d93025';
      } else if (inv.slaStatus === 'Future Dated') {
        slaBadge.innerText = '⚠️ FUTURE DATED';
        slaBadge.style.background = '#fef7e0'; // Light Yellow
        slaBadge.style.color = '#e37400';
      } else if (inv.slaStatus === 'Within SLA') {
        slaBadge.innerText = `${inv.processingDays} DAYS`;
        slaBadge.style.background = '#e6f4ea'; // Light Green
        slaBadge.style.color = '#1e8e3e';
      } else {
        slaBadge.innerText = '';
        slaBadge.style.background = 'transparent';
      }
      
      let displayTotal = Number(inv.totalAmount || 0);
      document.getElementById('invTotal').value = displayTotal.toLocaleString('en-IN', { minimumFractionDigits: 2 });
      
      document.getElementById('supplierLedger').value = inv.supplierLedger || inv.vendorName || '';

      // NEW: Populate the financial breakdown and GSTIN
      document.getElementById('invGstin').value = inv.gstin || 'Unregistered';
      document.getElementById('invTaxable').value = Number(inv.taxable || 0).toLocaleString('en-IN', { minimumFractionDigits: 2 });
      
      let totalGst = Number(inv.cgst || 0) + Number(inv.sgst || 0) + Number(inv.igst || 0) + Number(inv.cess || 0);
      document.getElementById('invGst').value = totalGst.toLocaleString('en-IN', { minimumFractionDigits: 2 });
      
      let chargesAndRoundoff = displayTotal - (Number(inv.taxable || 0) + totalGst);
      document.getElementById('invCharges').value = chargesAndRoundoff.toLocaleString('en-IN', { minimumFractionDigits: 2 });

      // Populate Items Table
      const tbody = document.getElementById('itemsTableBody');
      tbody.innerHTML = '';
      
      if (!inv.items || inv.items.length === 0) {
        tbody.innerHTML = '<tr><td colspan="5" style="text-align: center; color: #888; padding: 12px;">No line items found for this invoice.</td></tr>';
      } else {
        inv.items.forEach((item, idx) => {
          const lineTotal = Number(item.total || 0);

          const tr = document.createElement('tr');
          tr.innerHTML = `
            <td><b>${item.desc || ''}</b><br><small style="color:#5f6368">HSN: ${item.hsn || '-'} | Qty: ${item.qty || 1} ${item.unit || 'NOS'} @ ₹${Number(item.rate || 0).toFixed(2)}</small></td>
            <td>
              <select style="width:100%; padding:4px;" onchange="toggleLineType(${idx}, this.value)">
                <option value="Stock Item" ${item.lineType === 'Stock Item' ? 'selected' : ''}>Stock Item</option>
                <option value="Ledger / Expense" ${item.lineType === 'Ledger / Expense' ? 'selected' : ''}>Ledger / Expense</option>
                <option value="Service" ${item.lineType === 'Service' ? 'selected' : ''}>Service</option>
                <option value="Discount" ${item.lineType === 'Discount' ? 'selected' : ''}>Discount</option>
                <option value="Additional Charge" ${item.lineType === 'Additional Charge' ? 'selected' : ''}>Additional Charge</option>
                <option value="Round Off" ${item.lineType === 'Round Off' ? 'selected' : ''}>Round Off</option>
              </select>
            </td>
            <td>
              <input type="text" id="stock_${idx}" value="${item.stockItem || item.expenseHead || ''}" 
                     placeholder="Tally Ledger or Item Name" style="width:100%; padding:4px;">
            </td>
            <td style="text-align: right; font-weight: 500;">₹${lineTotal.toFixed(2)}</td>
          `;
          tbody.appendChild(tr);
        });
      }

      document.getElementById('prevBtn').disabled = currentIndex === 0;
      document.getElementById('nextBtn').disabled = currentIndex === filteredInvoices.length - 1;
    }

    function toggleLineType(idx, newType) {
      const inv = filteredInvoices[currentIndex];
      inv.items[idx].lineType = newType;
      const mappedInput = document.getElementById(`stock_${idx}`);

      if (newType === 'Ledger / Expense') {
        mappedInput.value = inv.items[idx].expenseHead || 'Purchase Account';
      } else {
        mappedInput.value = inv.items[idx].stockItem || inv.items[idx].desc || '';
      }
    }

    function recordDecision(action, reason = '') {
      const inv = filteredInvoices[currentIndex];
      const supplierLedger = document.getElementById('supplierLedger').value.trim();

      const updatedItems = (inv.items || []).map((it, idx) => ({
        itemRow: it.itemRow,
        desc: it.desc,
        lineType: it.lineType,
        stockItem: document.getElementById(`stock_${idx}`) ? document.getElementById(`stock_${idx}`).value.trim() : '',
        expenseHead: document.getElementById(`expense_${idx}`) ? document.getElementById(`expense_${idx}`).value.trim() : ''
      }));

      const dictKey = inv.invoiceNumber ? inv.invoiceNumber : ('row_' + inv.invRow);

      decisions[dictKey] = {
        invRow: inv.invRow,
        invoiceNumber: inv.invoiceNumber,
        vendorName: inv.vendorName,
        supplierLedger: supplierLedger || inv.vendorName,
        action: action,
        rejectionReason: reason,
        items: updatedItems
      };

      if (currentIndex < filteredInvoices.length - 1) {
        currentIndex++;
        renderCurrentInvoice();
      } else {
        renderCurrentInvoice();
        const toast = document.getElementById('customToast');
        toast.innerText = 'All loaded invoices reviewed. Click "Submit Decisions" to commit changes.';
        toast.style.display = 'block';
        setTimeout(() => toast.style.display = 'none', 3500);
      }
    }

    function navigate(delta) {
      const target = currentIndex + delta;
      if (target >= 0 && target < filteredInvoices.length) {
        currentIndex = target;
        renderCurrentInvoice();
      }
    }

    function openRejectModal() {
      document.getElementById('rejectionReasonInput').value = '';
      document.getElementById('rejectModal').style.display = 'flex';
    }

    function closeRejectModal() {
      document.getElementById('rejectModal').style.display = 'none';
    }

    function confirmReject() {
      const reason = document.getElementById('rejectionReasonInput').value.trim();
      if (!reason) {
        alert('Rejection reason is mandatory.');
        return;
      }
      closeRejectModal();
      recordDecision('Reject', reason);
    }

    function commitBatch() {
      const payload = Object.values(decisions);
      if (payload.length === 0) {
        alert('Please record at least one decision (Approve, Reject, or Skip) before submitting.');
        return;
      }

      const submitBtn = document.querySelector('.btn-primary');
      submitBtn.disabled = true;

      document.getElementById('contentContainer').style.opacity = '0.5';
      google.script.run
        .withSuccessHandler(res => {
          if (google.script && google.script.host && typeof google.script.host.close === 'function') {
            google.script.host.close();
          } else {
            document.body.innerHTML = '<h3 style="text-align:center; margin-top:50px;">Submissions recorded. You can safely close this tab.</h3>';
          }
        })
        .withFailureHandler(err => {
          alert('Submission error: ' + err);
          document.getElementById('contentContainer').style.opacity = '1';
          submitBtn.disabled = false;
        })
        .submitReviewBatch(payload);
    }
  </script>
